Ele sobe as duas portas (8999 e 9000), aplica o limite de 30 FPS para não travar o PC e usa a matemática do FFT. ```javascript const WebSocket = require('ws'); const portAudio = require('naudiodon'); const fft = require('fft-js').fft; const fftUtil = require('fft-js').util; console.log("=== Iniciando Sistema do Kraken V6 ==="); // --------------------------------------------------------- // 1. SERVIDOR WEBNOWPLAYING (Porta 8999) // --------------------------------------------------------- const wssWnp = new WebSocket.Server({ port: 8999 }); console.log("[+] Servidor WebNowPlaying aguardando na porta 8999..."); wssWnp.on('connection', ws => { ws.on('message', msg => { const msgStr = msg.toString(); wssWnp.clients.forEach(client => { if (client !== ws && client.readyState === WebSocket.OPEN) { client.send(msgStr); } }); }); ws.on('error', console.error); }); // --------------------------------------------------------- // 2. SERVIDOR EQUALIZADOR FFT (Porta 9000) // --------------------------------------------------------- const wssEq = new WebSocket.Server({ port: 9000 }); console.log("[+] Microserviço Equalizador aguardando na porta 9000..."); // Procura o dispositivo de Loopback (captura do desktop) const devices = portAudio.getDevices(); const targetDevice = devices.find(d => (d.name.toLowerCase().includes('misturador') || d.name.toLowerCase().includes('mix') || d.hostAPIName.includes('WASAPI')) && d.maxInputChannels > 0 ); console.log(`[EQ] Escutando áudio em: ${targetDevice ? targetDevice.name : 'Padrão do Sistema'}`); const ai = new portAudio.AudioIO({ inOptions: { channelCount: 1, // Transformado em Mono para facilitar o cálculo do FFT sampleFormat: portAudio.SampleFormat16Bit, sampleRate: 44100, deviceId: targetDevice ? targetDevice.id : -1, closeOnError: false } }); let lastSendTime = 0; const FPS_LIMIT = 1000 / 30; // Limita o envio via WebSocket para ~30 FPS ai.on('data', buf => { if (wssEq.clients.size === 0) return; // Não processa se o Kraken não estiver conectado const now = Date.now(); if (now - lastSendTime < FPS_LIMIT) return; lastSendTime = now; // O FFT exige que a amostra seja potência de 2 (ex: 512) const sampleSize = 512; if (buf.length < sampleSize * 2) return; let signal = []; for (let i = 0; i < sampleSize; i++) { // Converte os bytes do áudio bruto em valores decimais signal.push(buf.readInt16LE(i * 2) / 32768.0); } try { const phasors = fft(signal); const magnitudes = fftUtil.fftMag(phasors); // Transforma em frequências (Hz) // Função para calcular a média de uma faixa de frequências const getAverage = (startBin, endBin) => { let sum = 0; for (let i = startBin; i <= endBin; i++) sum += magnitudes[i]; return sum / (endBin - startBin + 1); }; // Divide o som em 5 faixas reais e aplica multiplicadores let b1 = getAverage(1, 4) * 8.0; // Graves (Bumbo) let b2 = getAverage(5, 12) * 12.0; // Médio-Graves let b3 = getAverage(13, 30) * 18.0;// Médios (Voz) let b4 = getAverage(31, 70) * 22.0;// Médio-Agudos let b5 = getAverage(71, 150) * 28.0;// Agudos (Pratos) const msg = `EQ:${b1.toFixed(2)},${b2.toFixed(2)},${b3.toFixed(2)},${b4.toFixed(2)},${b5.toFixed(2)}`; wssEq.clients.forEach(c => { if (c.readyState === WebSocket.OPEN) c.send(msg); }); } catch(e) { // Ignora erros silenciosos caso o buffer venha quebrado do Windows } }); ai.start(); ``` --- ### Passo 3: O Novo `Index15 com beat - Copia.html` Aqui, as tags do DOM agora são estáticas, acabando com a perda de performance, e a lógica da API do NZXT busca diretamente a RTX 4060, isolando o vídeo integrado no cálculo. ```html Tocando Agora - Kraken V6 Otimizado
